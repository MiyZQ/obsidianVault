
虚拟节点用于构建线性表，可以**统一头、尾、中间节点增删改查的边界逻辑**，双链表添加虚拟头尾节点、单链表添加虚拟头节点

链表边界逻辑：
1. 是否存在元素 $0\leq index\lt self.size$
2. 是否可添加元素 $0\leq index\leq self.size$

下面为双链表示例
```python
from functools import wraps

class Node:
    def __init__(self, val=None, prev=None, next=None):
        self.val = val
        self.next = next
        self.prev = prev

class MyLinkedList:
    def __init__(self):
        self.head = Node()
        self.tail = Node(prev=self.head)
        self.head.next = self.tail
        self._size = 0

    # other tool function
    
    def size(self):
        return self._size

    def is_empty(self):
        return self._size == 0

    def is_element_index(self, index):
        return 0 <= index < self._size

    def is_position_index(self, index):
        return 0 <= index <= self._size

    @staticmethod
    def check_index(is_position=False):
        def outer(fn):
        	@wraps(fn)
            def wrapper(inst, index, *args, **kwargs):
            flag = inst.is_position_index(index) if is_position else inst.is_element_index(index)
                if not flag:
                    raise IndexError(f"Index: {index}, Size: {inst._size}")
                return fn(inst, index, *args, **kwargs)
            return wrapper
        return outer

    def __str__(self):
        res = []
        node = self.head.next
        while node != self.tail:
            res.append(str(node.val))
            node = node.next
        return "[" + ",".join(res) + "]"

    # method

    def add(self, index, val):
        prev = self.get_position(index)
        next_node = prev.next
        node_new = Node(val, prev=prev, next=next_node)
        prev.next = node_new
        next_node.prev = node_new
        self._size += 1

    def add_tail(self, val):
        self.add(self._size, val)

    def add_head(self, val):
        self.add(0, val)

    def remove(self, index):
        temp = self.get_element(index)
        prev = temp.prev
        next_node = temp.next
        prev.next = next_node
        next_node.prev = prev
        self._size -= 1
        return temp.val

    def remove_head(self):
        return self.remove(0)

    def remove_tail(self):
        return self.remove(self._size - 1)

    def set(self, index, val):
        node = self.get_element(index)
        val_old = node.val
        node.val = val
        return val_old

    def _get(self, index, begin, end, is_position) -> Node:
        if index < self._size // 2:
            node = begin
            while index:
                node = node.next
                index -= 1
        else:
            node = end
            index = self._size - index if is_position else self._size - 1 - index
            while index:
                node = node.prev
                index -= 1
        return node

    @check_index()
    def get_element(self, index) -> Node:
        return self._get(index, self.head.next, self.tail.prev, is_position=False)

    @check_index(is_position=True)
    def get_position(self, index) -> Node:
        return self._get(index, self.head, self.tail.prev, is_position=True)
```
