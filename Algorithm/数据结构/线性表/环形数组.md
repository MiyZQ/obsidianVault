
环形数组即索引对长度取模，在 $[start, end)$ 区间内包含数组元素，用双指针来动态扩容，可以做到 **O(1) 时间在头部增删元素**，用来==实现队列==

> [!NOTE] Python 中的双端队列
> ```python
> from collections import deque
> 
> lst = deque()
> ```

环形数组实现：
```python
from functools import wraps


class CycleArray:
    def __init__(self, _capacity=1) -> None:
        self._capacity = _capacity
        self._size = 0

        self.arr = [None] * _capacity

        self.start = 0
        self.end = 0

    # other tool function

    def is_full(self):
        return self._size == self._capacity

    def capacity(self):
        return self._capacity

    def size(self):
        return self._size

    def is_empty(self):
        return self._size == 0

    def is_element_index(self, index):
        return 0 <= index < self._size

    @staticmethod
    def check_element_index(fn):
        @wraps(fn)
        def wrapper(inst, index, *args, **kwargs):
            if not inst.is_element_index(index):
                raise IndexError(f"Index: {index}, Count: {inst._size}")
            return fn(inst, index, *args, **kwargs)

        return wrapper

    def resize(self, size_new):
        arr_new = [None] * size_new

        for i in range(self._size):
            arr_new[i] = self.arr[(self.start + i) % self._capacity]
        self.arr = arr_new

        self.start = 0
        self.end = self._size
        self._capacity = size_new

    @staticmethod
    def resize_auto(is_add=False):
        def outer(fn):
            @wraps(fn)
            def wrapper(inst, *args, **kwargs):
                if is_add and inst.is_full():
                    inst.resize(inst._capacity * 2)

                elif not is_add and inst.is_empty():
                    raise IndexError("Array is empty")

                res = fn(inst, *args, **kwargs)

                if not is_add and 0 < inst._size <= inst._capacity // 4:
                    inst.resize(max(1, inst._capacity // 2))

                return res

            return wrapper

        return outer

    # method

    @resize_auto(is_add=True)
    def add_first(self, val):
        self.start = (self.start - 1 + self._capacity) % self._capacity
        self.arr[self.start] = val
        self._size += 1

    @resize_auto(is_add=True)
    def add_last(self, val):
        self.arr[self.end] = val
        self.end = (self.end + 1) % self._capacity
        self._size += 1

    @resize_auto()
    def remove_first(self):
        val = self.arr[self.start]
        self.arr[self.start] = None

        self.start = (self.start + 1) % self._capacity

        self._size -= 1
        return val

    @resize_auto()
    def remove_last(self):
        self.end = (self.end - 1 + self._capacity) % self._capacity

        val = self.arr[self.end]
        self.arr[self.end] = None

        self._size -= 1
        return val

    @check_element_index
    def get(self, index):
        return self.arr[(self.start + index) % self._capacity]
```
