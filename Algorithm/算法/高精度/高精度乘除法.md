
乘法，其中针对前导 0 进行了输出过滤处理：
```cpp
void mul(string a,string b) {
    vector<int> res(a.size() + b.size());

    reverse(a.begin(),a.end());
    reverse(b.begin(),b.end());

    if (a.size() < b.size()) a.swap(b);

    for (int i{0}; i < b.size(); ++i) {
        int carry{0};
        int j{0};
        for (;j < a.size();++j) {
            int tmp = (a[j] - '0') * (b[i] - '0') + carry;
            carry = tmp / 10;

            res[i+j] += tmp % 10;
            res[i+j+1] += res[i+j] / 10;
            res[i+j] %= 10;
        }
        if (carry > 0) res[i+j] += carry;
    }

    bool flag = true;
    for (int i(a.size()+b.size()-1);i>=0;--i) {
        if (i>0 && flag && res[i] == 0) continue;
        if (flag && res[i] != 0) flag = false;
        cout<<res[i];
    }
}
```
