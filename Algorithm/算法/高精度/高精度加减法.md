
加法，主要思想是用 carry 存储进位数。下面算法前提是输入不带前导 0 如 000021 这样的数，或者补充这样的逻辑
```cpp
vector<int> add(string a,string b) {
    vector<int> res;
    int length = max(a.size(),b.size());
    res.reserve(length+1);

    reverse(a.begin(),a.end());
    reverse(b.begin(),b.end());

    int carry{0};
    for (int i{0}; i < length; ++i) {
        int tmp{carry};
        if (i < a.size()) tmp += a[i] - '0';
        if (i < b.size()) tmp += b[i] - '0';
        res.push_back(tmp%10);
        carry = tmp/10;
    }
    if (carry) {
        res.push_back(carry);
    }
    reverse(res.begin(),res.end());
    return res;
}
```

减法类似于加法算法，需要特判为 0 的情况，因为 .size() 返回类型为无符号，若为 0 再减 1 即 res.size() - 1会数值溢出
```cpp
bool isMax(string& a,string& b) {
    if (a.size()==b.size()) {
        for (int i(a.size());i>=0;--i) {
            if (a[i]!=b[i]) {
                return a[i] > b[i];
            }
        }
        return true;
    }
    else {
        return a.size() > b.size();
    }
}

void to_minus(string a,string b) {
    vector<int> res;

    reverse(a.begin(),a.end());
    reverse(b.begin(),b.end());

    bool flag = isMax(a,b);
    if (!flag) {
        a.swap(b);
    }

    int length = a.size();
    res.reserve(length);

    int carry{0};
    for (int i{0}; i < length; ++i) {
        int tmp{-carry};
        if (i < a.size()) tmp += a[i] - '0';
        if (i < b.size()) tmp -= b[i] - '0';
        
        if (tmp < 0) {
            carry = 1;
            tmp += 10;
        }
        else {
            carry = 0;
        }
        res.push_back(tmp%10);
    }
    while (res.size() > 1 && res.back() == 0) {
        res.pop_back();
    }
    
    if (res.size() == 1 && res[0] == 0) {
        cout<<0<<endl;
    }
    else {
        if (!flag) cout<<"-";
        for (int i(res.size()-1);i>=0;--i) {
            cout<<res[i];
        } 
        cout<<endl;
    }
}
```
