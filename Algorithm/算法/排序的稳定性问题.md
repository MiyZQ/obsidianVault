
> cpp 中的 sort 是不稳定排序，当给数组排序时要求类稳定性时，可以附加一个数据 idx 指定其元素索引，将 idx 加入到 cmp 函数中的比较就可做到稳定性效果

> [!NOTE] 洛谷 P1104
> 输出共有 nn 行，即 nn 个年龄从大到小同学的姓名（如果有两个同学年龄相同，输入靠后的同学先输出）

```cpp
#include<bits/stdc++.h>
using namespace std;
struct Stu{
	string name;
	int y,m,d;
	int idx; // 索引信息
} stu[105];
auto cmp(const Stu& a,const Stu&b) -> bool {
	if (a.y!=b.y){
		return a.y<b.y;
	}
	else{
		if (a.m!=b.m){
			return a.m<b.m;
		}
		else{
			if(a.d!=b.d){
				return a.d<b.d;
			}
			else return a.idx>b.idx; // 加入类稳定性的比较
		}
	}
}
int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n;
	cin>>n;
	for(int i=0;i<n;++i){
		cin>>stu[i].name>>stu[i].y>>stu[i].m>>stu[i].d;
		stu[i].idx = i;
	}
	sort(stu,stu+n,cmp);
	for(int i=0;i<n;++i){
		cout<<stu[i].name<<'\n';
	}
    return 0;
}
```
