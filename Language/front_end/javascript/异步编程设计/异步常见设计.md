
#### Promise 对象间运算
两个解析为数字的 Promise 对象间运算，新的 Promise 对象解析为其运算结果
```js
var addTwoPromises = async function(promise1, promise2) {
    return Promise.all([promise1,promise2])
        .then(([a,b])=>a+b)
}
/**
 * addTwoPromises(Promise.resolve(2), Promise.resolve(2))
 *   .then(console.log); // 4
 */
```

#### 休眠函数
即休眠一定时间后解析任何值的函数
```js
async function sleep(millis) {
    return new Promise(resolve=>setTimeout(resolve,millis))
}
/**
 * let t = Date.now()
 * sleep(100).then(() => console.log(Date.now() - t)) // 100
 */
```

#### 可取消的延迟函数
即运行后经过一定延时后不调用返回的取消函数，则内部将调用接收的参数函数
```js
var cancellable = function(fn, args, t) {
    const times = setTimeout(()=>fn(...args),t)
    return ()=>clearTimeout(times)
}
/**
 * cancel = cancellable(console.log,"hello world",3000)
 * setTimeout(cancel,2999)
 */
```

#### 限时函数
接收异步函数返回限时函数，异步函数完成则返回限时结果，否则拒绝
```js
function timeLimit(fn, t) {
    return async function (...args) {
        let timer
        const timeoutPromise = new Promise((res, rej) => {
            timer = setTimeout(() => rej('Time Limit Exceeded'), t)
        })

        return Promise.race([fn(...args), timeoutPromise])
            .then(res => {
                clearTimeout(timer)
                return res
            })
    }
}
```

#### 函数防抖
即延迟 t 执行的函数，在时间窗口内再次调用将取消
```js
const log = debounce(console.log, 100);

log('Hello'); // cancelled
log('Hello'); // cancelled
log('Hello'); // Logged at t=100ms
```

实现：
```js
var debounce = function(fn, t) {
    let timer = null
    return function() {
        clearTimeout(timer)
        timer = setTimeout(()=>fn(...arguments),t)
    }
}
```
