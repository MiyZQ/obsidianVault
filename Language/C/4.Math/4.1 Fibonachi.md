
Fibonachi数列由递推定义：
$$F(0)=F(1)=1;$$
$$F(n)=F(n-1)+F(n-2);$$

一般采用矩阵快速幂的方法求解，以下为示意：
```c
typedef long long ll;  
typedef struct{  
    ll mo[2][2];  
}matrix;  
matrix E ={1,0,0,1};  
matrix mul(matrix a,matrix b){  
    matrix re = {};  
    for(int i=0;i<2;i++) for(int j=0;j<2;j++) for(int k=0;k<2;k++)  
                re.mo[i][j] += a.mo[i][k]*b.mo[k][j];  
    return re;  
}  
matrix opp(matrix a,int n){  
    matrix re = E;  
    while (n>0){  
        if(n&1) re = mul(re,a);  
        a = mul(a,a);  
        n >>=1;  
    }  
    return re;  
}  
ll F(int n){  
    matrix a = {1,1,1,0};  
    return opp(a,n).mo[1][0];  
}
```
