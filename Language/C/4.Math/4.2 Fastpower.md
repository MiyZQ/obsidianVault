
快速幂的时间复杂度为O(logn)，空间复杂度为O(1)，通过二分思想减少取幂运算次数
关于快速乘法可见[4.9 Prime](4.9%20Prime.md)

## 基础快速幂：计算 $a^n$
```c
long long fp(long long a, long long n) {  
    long long result = 1;  
    while (n > 0) {  
        if (n & 1) result *= a;  
        a *= a;  
        n >>= 1;}  
    return result;  
}
```

## 带模运算的快速幂：计算 $a^{n}$mod m
```c
long long mfp(long long a, long long n, long long m) {  
    long long result = 1;  
    a %= m;  
    while (n > 0) {  
        if (n & 1) result = (result * a) % m;  
        a = (a * a) % m;  
        n >>= 1;  
    }  
    return result;  
}
```

## 计算模逆元

这里要求m为素数，计算机上一般使用1e9+7,备选998244353
```c
long long mod_inverse(long long a, long long m) {  
    return mfp(a, m - 2, m);  
}
```

## 计算组合数$\dbinom{n}{k}$
```c
long long combination_mod(long long n, long long k, long long m) {  
    if (k < 0 || k > n) return 0;  
    if (k == 0 || k == n) return 1;  
    // 使用公式: C(n, k) = n! / (k! * (n-k)!)  
    // 预处理阶乘和逆元可以获得更好性能  
    long long a = 1;  
    long long b = 1;  
    for (long long i = 1; i <= k; i++) {  
        a = (a * (n - i + 1)) % m;  
        b = (b * i) % m;  
    }  
    return (a * mod_inverse(b, m)) % m;  
}
```
