
以下举出生成全排列的算法

## 不去重递归生成全排列，复杂度O(n!),空间O(n)
```c
void permute(char *str, int left, int right) {  
    if (left == right) printf("%s\n", str);  
    else {  
        for (int i = left; i <= right; i++) {  
            swap(str[left], str[i])  
            permute(str, left + 1, right);  
            swap(str[left], str[i])  
        }   
    }  
}
```

## 去重字典序生成全排列，复杂度O(n\*n!),空间O(1)
```c
int compare(const void *a, const void *b) {  
    return (*(char *)a - *(char *)b);  
}  
void reverse(char *str, int start, int end) {  
    while (start < end) {  
        swap(str[start], str[end])  
        start++;  
        end--;  
    }  
}  
int nextPermutation(char *str, int n) {  
    int i = n - 2;  // 从右往左找第一个递减的位置
    while (i >= 0 && str[i] >= str[i + 1]) i--;  
    if (i < 0) return 0;  
    int j = n - 1;  // 从右往左找第一个大于str[i]的字符 
    while (j > i && str[j] <= str[i]) j--;  
    swap(str[i], str[j])  
    reverse(str, i + 1, n - 1);  
    return 1;  
}  
void permute(char *str,int n){  
    qsort(str, n, sizeof(char), compare);  
    do printf("%s\n", str);  
    while (nextPermutation(str, n));  
}
```
