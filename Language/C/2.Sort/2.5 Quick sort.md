
快速排序为不稳定排序，复杂度O(nlogn)，最坏情况\[正序或倒序]是$O(n^2)$
```c
void quicksort(int arr[],int left,int right)  
{  
    if (left >= right) return;  
    int mid = left + (right - left) / 2; 
    if (arr[left] > arr[mid]) swapi(arr[left], arr[mid])  
    if (arr[left] > arr[right]) swapi(arr[left], arr[right])  
    if (arr[mid] > arr[right]) swapi(arr[mid], arr[right])  
    swapi(arr[left], arr[mid])  // 中间值移到 left 作为基准  
    int base = arr[left],l = left,r = right;//根据情况基准选择前，后，中的一个,最好取中  
    while (l<r)  
    {  
        while (l < r && arr[r] >= base) r--;  
        arr[l] = arr[r];  
        while (l < r && arr[l] <= base) l++;  
        arr[r] = arr[l];  
    }  
    arr[r] = base;  //最后指针r，l指向同一位置  
    quicksort(arr,left,r-1);  
    quicksort(arr,r+1,right);  
}
```
如果省事base也可直接随便取一项

迭代版本：
```c
typedef struct{
	int start,end;
}range;

range newrange( int s, int e){
	range r = {s,e};
	return r;
}

void quicksort( int* arr, const int len){
	if(len<=0) return;
	range r[len];
	int p = 0;
	r[p++] = newrange(0,len-1);
	while (p>0) {
		range ra = r[--p];
		if(ra.start >= ra.end) continue;
		int mid = arr[(ra.start+ra.end)/2];
		int l = ra.start, r = ra.end;
		do {
			while (arr[l]<mid) l++;
			while (arr[r]>mid) r--;
			if(l <= r){
				swapi(arr[l],arr[r])
				l++;
				r--;
			}
		} while (l <= r)
		if (ra.start < r) r[p++] = newrange(ra.start,r);
		if (ra.end > l) r[p++] = newrange(l,ra.end);
	}
}
```
