
堆排序的复杂度稳定在O(nlogn)，为原地排序，空间占用O(1)，是不稳定排序，对缓存不友好

## 递归堆调整
```c
void heapify(int arr[], int n, int i) {  
    int max = i, left = 2 * i + 1, right = left + 1;  
    if (left < n && arr[left] > arr[max]) max = left;  
    if (right < n && arr[right] > arr[max]) max = right;  
    if (max != i) {  
        swapi(arr[i], arr[max])  
        heapify(arr, n, max);  
    }  
}
```

## 排序函数
```c
void heapSort(int arr[], int n) {  
    // 构建最大堆（从最后一个非叶子节点开始）  
    for (int i = n / 2 - 1; i >= 0; i--) heapify(arr, n, i);  
    // 逐个提取最大值到数组末尾  
    for (int i = n - 1; i > 0; i--) {  
        swapi(arr[0], arr[i])  
        heapify(arr, i, 0);  
    }  
}
```

## 优化的堆调整，提速用
```c
void heapify(int arr[], int n, int i) {  
    int current = i;  
    while (1) {  
        int left = 2 * current + 1;  
        int right = left + 1;  
        int max = current;  
        if (left < n && arr[left] > arr[max]) max = left;  
        if (right < n && arr[right] > arr[max]) max = right;  
        if (max == current) break;  
        swapi(arr[current], arr[max])  
        current = max;  
    }  
}
```
