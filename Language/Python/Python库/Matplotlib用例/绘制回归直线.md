
> [!NOTE]
> np.polyfit(x,y,1) 返回线性回归直线参数 k,b
> 更一般的，np.polyfit(x,y,n) 返回 多项式拟合的参数列表

```python
import matplotlib.pyplot as plt
import numpy as np
plt.rcParams["font.family"] = ["Microsoft YaHei"]

x1 = np.linspace(2,4,5)
y1 = np.array([20.1,25.2,30.2,35.2,40.3])

x2 = np.linspace(50,250,5)
y2 = np.array([13,27.5,43.2,56.7,70.7])

x3 = np.linspace(-2,2,5)
y3 = np.array([-13.2,-6.7,0,6.7,13.4])

plt.subplot(2,2,1)
k1, b1 = np.polyfit(x1,y1,1)
yy1 = k1 * x1 + b1
plt.xlabel("I_H/mA")
plt.ylabel("U_H/mV")
plt.title(f'斜率：{k1}')
plt.plot(x1,y1,'ro')
plt.plot(x1,yy1,'b-')

plt.subplot(2,2,2)
k2, b2 = np.polyfit(x2,y2,1)
yy2 = k2 * x2 + b2
plt.xlabel("B/mT")
plt.ylabel("U_H/mV")
plt.title(f'斜率：{k2}')
plt.plot(x2,y2,'ro')
plt.plot(x2,yy2,'b-')

plt.subplot(2,2,3)
k3, b3 = np.polyfit(x3,y3,1)
yy3 = k3 * x3 + b3
plt.xlabel("r/mm")
plt.ylabel("U_H/mV")
plt.title(f'斜率：{k3}')
plt.plot(x3,y3,'ro')
plt.plot(x3,yy3,'b-')

plt.tight_layout()
plt.show()
```
