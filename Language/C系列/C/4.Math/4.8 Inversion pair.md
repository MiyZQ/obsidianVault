
该算法求解数组逆序对的个数，采用分治算法，即merge排序过程，复杂度O(nlogn),空间O(n)
```c
int inversions(const int *p,int n){  
    if(n<=1) return 0;  
    int mid = n/2;  
    int *left = (int*)malloc(mid*sizeof(int));  
    int *right = (int*)malloc((n-mid)*sizeof(int));  
    int *sort = (int*)malloc(n*sizeof(int));  
    memcpy(left,p,mid*sizeof(int));  
    memcpy(right,p+mid,(n-mid)*sizeof(int));  
    int inv = inversions(left,mid)+ inversions(right,n-mid); 
    int i=0,j=0,k=0;  
    while (i<mid && j<n-mid){  
        if(left[i] <= right[j]) sort[k++] = left[i++];  
        else{  
            sort[k++] = right[j++];  
            inv += mid-i;  
        }  
    }  
    while (i<mid) sort[k++] = left[i++];  
    while (j<n-mid) sort[k++] = right[j++];  
    free(left);  
    free(right);  
    free(sort);  
    return inv;  
}
```
在递归中left、right均排好序，从而简化了inv加法次数
