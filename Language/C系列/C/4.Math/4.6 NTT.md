
快速数论变换相比FFT更适合于高精度乘法运算，复杂度为O(nlogn)，同样处理卷积问题

## 数学推导

设$p$为质数，$n$为2的幂次且满足$n \mid (p-1)$。由原根定理，存在原根$g$模$p$，定义$n$次单位根：
$$\omega = g^{(p-1)/n} \mod p$$
满足：
$$\omega^n \equiv 1 \mod p$$$$\omega^{n/2} \equiv -1 \mod p  $$
$$\omega^{k} \neq \omega^{l}对 0 \leq k < l < n$$
$$\sum_{j=0}^{n-1} \omega^{jk} \equiv \begin{cases} n & \text{若 } k \equiv 0 \mod n \\ 0 & \text{否则} \end{cases} \mod p$$
NTT：
$$\hat{a}_k = \sum_{j=0}^{n-1} a_j \omega^{jk} \mod p, \quad k = 0,1,\ldots,n-1$$
NTT可表示为矩阵乘法：
$$\begin{bmatrix} \hat{a}_0 \\ \hat{a}_1 \\ \vdots \\ \hat{a}_{n-1} \end{bmatrix} = \begin{bmatrix} \omega^{0} & \omega^{0} & \cdots & \omega^{0} \\ \omega^{0} & \omega^{1} & \cdots & \omega^{n-1} \\ \vdots & \vdots & \ddots & \vdots \\ \omega^{0} & \omega^{n-1} & \cdots & \omega^{(n-1)^2} \end{bmatrix} \begin{bmatrix} a_0 \\ a_1 \\ \vdots \\ a_{n-1} \end{bmatrix}$$
卷积定理：
$$c_m = \sum_{i+j \equiv m \mod n} a_i b_j$$
$$\hat{c}_k = \hat{a}_k \cdot \hat{b}_k \mod p$$
对于循环卷积，这正是所需结果。对于线性卷积，需通过补零扩展到$2n$点。

将多项式按奇偶下标分治：
$$A(x) = A_0(x^2) + x A_1(x^2)$$
$$A_0(x) = \sum_{j=0}^{n/2-1} a_{2j} x^j, \quad A_1(x) = \sum_{j=0}^{n/2-1} a_{2j+1} x^j$$
对于$k < n/2$：
$$\hat{a}_k = \sum_{j=0}^{n/2-1} a_{2j} \omega^{2jk} + \omega^k \sum_{j=0}^{n/2-1} a_{2j+1} \omega^{2jk} = \hat{a}_{0,k} + \omega^k \hat{a}_{1,k}$$
对于$k \geq n/2$，令$k' = k - n/2$：
$$\hat{a}_k = \hat{a}_{0,k'} + \omega^{k'} \omega^{n/2} \hat{a}_{1,k'} = \hat{a}_{0,k'} - \omega^{k'} \hat{a}_{1,k'}$$
设$n = 2^m$，迭代过程：
$$\begin{align*} &\text{对于} L = 2,4,8,\ldots,n: \\ &\quad \text{对于每组长度为} L \text{的块}: \\ &\quad \quad \omega_L = \omega^{(p-1)/L} \\ &\quad \quad \text{对于} j = 0,1,\ldots,L/2-1: \\ &\quad \quad \quad u = a[i+j] \\ &\quad \quad \quad v = \omega_L^j \cdot a[i+j+L/2] \\ &\quad \quad \quad a[i+j] = u + v \\ &\quad \quad \quad a[i+j+L/2] = u - v \end{align*}$$

==Cooley-Tukey算法==
```c
void ntt(long long *a, int n, int inv) {  
    int j = 0;  
    for (int i = 1; i < n; i++) {  
        int bit = n >> 1;  
        while (j >= bit) {  
            j -= bit;  
            bit >>= 1;  
        }  
        j += bit;  
        if (i < j) {  
            long long temp = a[i];  
            a[i] = a[j];  
            a[j] = temp;  
        }  
    }  
  
    for (int L = 2; L <= n; L <<= 1) {  
        long long wlen = mfp(G, (MOD - 1) / L, MOD);  
        if (inv) wlen = mfp(wlen, MOD - 2, MOD);    
  
        for (int i = 0; i < n; i += L) {  
            long long w = 1;  
            for (int j = 0; j < L / 2; j++) {  
                long long u = a[i + j];  
                long long v = (w * a[i + j + L / 2]) % MOD;  
                a[i + j] = (u + v) % MOD;                    // 蝴蝶操作  
                a[i + j + L / 2] = (u - v + MOD) % MOD;  
                w = (w * wlen) % MOD;  
            }  
        }  
    }  
    if (inv) {  
        long long inv_n = mfp(n, MOD - 2, MOD);  
        for (int i = 0; i < n; i++) {  
            a[i] = (a[i] * inv_n) % MOD;  
        }  
    }  
}
```

## 卷积
```c
void multi(long long *a, int n1, long long *b, int n2, long long *c) {  
    int n = 1;  
    int len = n1 + n2 - 1;  
    while (n < len) n <<= 1;  
      
    long long *a_ext = (long long *)calloc(n, sizeof(long long));  
    long long *b_ext = (long long *)calloc(n, sizeof(long long));  
    memcpy(a_ext, a, n1 * sizeof(long long));  
    memcpy(b_ext, b, n2 * sizeof(long long));  
      
    ntt(a_ext, n, 0);  
    ntt(b_ext, n, 0);  
    for (int i = 0; i < n; i++) a_ext[i] = (a_ext[i] * b_ext[i]) % MOD;  
    ntt(a_ext, n, 1);  
      
    memcpy(c, a_ext, len * sizeof(long long));  
    free(a_ext);  
    free(b_ext);  
}
```

## 高精度乘法
```c
char* bigint_multiply_ntt(const char *num1, const char *num2) {  
    int len1 = strlen(num1);  
    int len2 = strlen(num2);  
    int len = len1 + len2;  
    int n = 1;  
    while (n < len) n <<= 1;  
    long long *a = (long long *)calloc(n, sizeof(long long));  
    long long *b = (long long *)calloc(n, sizeof(long long));  
    for (int i = 0; i < len1; i++) a[i] = num1[len1 - 1 - i] - '0';  
    for (int i = 0; i < len2; i++) b[i] = num2[len2 - 1 - i] - '0';  
      
    ntt(a, n, 0);  
    ntt(b, n, 0);  
    for (int i = 0; i < n; i++) a[i] = (a[i] * b[i]) % MOD;  
    ntt(a, n, 1);  
  
    long long *result = (long long *)malloc(n * sizeof(long long));  
    memcpy(result, a, n * sizeof(long long));  
    for (int i = 0; i < n - 1; i++) {  
        result[i + 1] += result[i] / 10;  
        result[i] %= 10;  
    }  
    int end = n - 1;  // 找到最高非零位  
    while (end > 0 && result[end] == 0) end--;  
      
    char *final = (char *)malloc((end + 2) * sizeof(char));    // 转换为字符串  
    for (int i = 0; i <= end; i++) final[i] = result[end - i] + '0';  
    final[end + 1] = '\0';  
    free(a);  
    free(b);  
    free(result);  
    return final;  
}
```
