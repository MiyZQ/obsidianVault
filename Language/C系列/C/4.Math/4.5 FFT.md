快速傅里叶变换的复杂度为O(nlogn),可以处理卷积、高精度乘法等运算

## 数学推导

对于序列$x[n]$，$n=0,1,\ldots,N-1$：
$$X[k] = \sum_{n=0}^{N-1} x[n] \cdot e^{-j\frac{2\pi kn}{N}}, \quad k = 0, 1, \ldots, N-1$$
定义旋转因子：$$W_N = e^{-j\frac{2\pi}{N}}$$
简写为：$$X[k] = \sum_{n=0}^{N-1} x[n] \cdot W_N^{kn}$$
假设$N=2^m$，将序列分解：
$$\begin{aligned} X[k] &= \sum_{n=0}^{N-1} x[n] \cdot W_N^{kn} \\ &= \sum_{r=0}^{N/2-1} x[2r] \cdot W_N^{2rk} + \sum_{r=0}^{N/2-1} x[2r+1] \cdot W_N^{(2r+1)k} \end{aligned}$$​​
利用旋转因子的周期性和对称性：
$$W_N^{2rk} = (W_N^2)^{rk} = W_{N/2}^{rk}$$
$$W_N^{(2r+1)k} = W_N^{2rk} \cdot W_N^k = W_{N/2}^{rk} \cdot W_N^k$$
代入原式：
$$\begin{aligned} X[k] &= \sum_{r=0}^{N/2-1} x[2r] \cdot W_{N/2}^{rk} + W_N^k \sum_{r=0}^{N/2-1} x[2r+1] \cdot W_{N/2}^{rk} \end{aligned}$$
定义偶数和奇数部分的DFT：
$$G[k] = \sum_{r=0}^{N/2-1} x[2r] \cdot W_{N/2}^{rk}, \quad k = 0, 1, \ldots, N/2-1$$
$$H[k] = \sum_{r=0}^{N/2-1} x[2r+1] \cdot W_{N/2}^{rk}, \quad k = 0, 1, \ldots, N/2-1$$
则原DFT可表示为：
$$X[k] = G[k] + W_N^k H[k], \quad k = 0, 1, \ldots, N/2-1$$
由于$G[k]$和$H[k]$的周期为$N/2$且：
$$W_N^{k + N/2} = W_N^k \cdot W_N^{N/2} = W_N^k \cdot e^{-j\pi} = -W_N^k$$
因此：
$$X[k + N/2] = G[k] - W_N^k H[k], \quad k = 0, 1, \ldots, N/2-1$$
FFT的核心是蝶形运算：
$$\begin{aligned} X[k] &= G[k] + W_N^k H[k] \\ X[k + N/2] &= G[k] - W_N^k H[k] \end{aligned}$$
FFT的复杂度递归关系：
$$T(N) = 2T\left(\frac{N}{2}\right) + O(N)$$
求解：
$$\begin{aligned} T(N) &= 2\left[2T\left(\frac{N}{4}\right) + O\left(\frac{N}{2}\right)\right] + O(N) \\ &= 4T\left(\frac{N}{4}\right) + 2O(N) \\ &= 2^{\log_2 N}T(1) + \log_2 N \cdot O(N) \\ &= O(N \log N) \end{aligned}$$

```c
void bit_reverse(complex double *x, int n) {  
    for (int i = 0, j = 0; i < n; i++) {  
        if (i < j) {  
            complex double temp = x[i];  
            x[i] = x[j];  
            x[j] = temp;  
        }  
        for (int k = n/2; (j ^= k) < k; k >>= 1);  
    }  
}
```

==迭代FFT/IFFT==
```c
void fft(complex double *x, int n, int inverse) {  
    bit_reverse(x, n);  
    for (int s = 2; s <= n; s <<= 1) {  
        double angle = 2*PI/s * (inverse ? -1 : 1);  
        complex double wn = cos(angle) + I*sin(angle);  
        for (int k = 0; k < n; k += s) {  
            complex double w = 1;  
            for (int j = 0; j < s/2; j++) {  
                complex double t = w * x[k + j + s/2];  
                complex double u = x[k + j];  
                x[k + j] = u + t;  
                x[k + j + s/2] = u - t;  
                w *= wn;  
            }  
        }  
    }  
    if (inverse) {  
        for (int i = 0; i < n; i++)  
            x[i] /= n;  
    }  
}
```
inverse为1表示反演IFFT

```c
#define PI 3.14159265358979323846
void fft(complex double x[], int n, int inverse) {  
    for (int i = 1, j = 0; i < n; i++) {  
        int bit = n >> 1;  
        for (; j >= bit; bit >>= 1) j -= bit;  
        j += bit;  
        if (i < j) {  
            complex double temp = x[i];  
            x[i] = x[j];  
            x[j] = temp;  
        }  
    }  
  
    for (int len = 2; len <= n; len <<= 1) {  
        double angle = (inverse ? 2.0 : -2.0) * PI / len;  
        complex double wlen = cexp(I * angle);  
        for (int i = 0; i < n; i += len) {  
            complex double w = 1.0;  
            for (int j = 0; j < len/2; j++) {  
                complex double u = x[i + j];  
                complex double v = w * x[i + j + len/2];  
                x[i + j] = u + v;  
                x[i + j + len/2] = u - v;  
                w *= wlen;  
            }  
        }  
    }  
    if (inverse) for (int i = 0; i < n; i++) x[i] /= n;  
}
```

## 高精度乘法

```c
char* multiply(const char *a, const char *b) {  
    int len_a = strlen(a), len_b = strlen(b);  
    int n = 1;  
    while (n < len_a + len_b) n <<= 1;  
    complex double *c = calloc(n, sizeof(complex double));  
    for (int i = 0; i < n; i++) {  
        double real = (i < len_a) ? (a[len_a-1-i]-'0') : 0;  
        double imag = (i < len_b) ? (b[len_b-1-i]-'0') : 0;  
        c[i] = real + I*imag;  
    }  
    fft(c, n, 0);      // FFT  
    for (int i=0; i<n; i++)  
        c[i] *= c[i];  
    fft(c, n, 1);      // IFFT  
    int *res = calloc(n, sizeof(int));  
    for (int i=0; i<n; i++)  
        res[i] = (int)round(cimag(c[i])/2); // 取虚部除以2  
    int carry = 0;  // 进位处理 
    for (int i=0; i<n; i++) {  
        res[i] += carry;  
        carry = res[i]/10;  
        res[i] %= 10;  
    }  
    int pos = n-1;  
    while (pos>0 && res[pos]==0) pos--;  
    char *result = calloc(pos+2, sizeof(char));  
    for (int i=0; i<=pos; i++)  
        result[i] = res[pos-i] + '0';  
    result[pos+1] = '\0';  
    free(c); free(res);  
    return result;  
}
```

```c
int convolution(const double x[], int len_x,  
                const double h[], int len_h,  
                double result[]) {  
    int len = len_x + len_h - 1;  
    int size = 1;  
    while (size < len) size <<= 1;  
  
    complex double *x_padded = calloc(size, sizeof(complex double));  
    complex double *h_padded = calloc(size, sizeof(complex double));  
    complex double *y_freq = calloc(size, sizeof(complex double));  
  
    for (int i = 0; i < len_x; i++) x_padded[i] = x[i];  
    for (int i = 0; i < len_h; i++) h_padded[i] = h[i];  
  
    fft(x_padded, size, 0);  
    fft(h_padded, size, 0);  
  
    for (int i = 0; i < size; i++) {  
        y_freq[i] = x_padded[i] * h_padded[i];  
    }  
  
    fft(y_freq, size, 1);  
  
    for (int i = 0; i < len; i++) {  
        result[i] = creal(y_freq[i]);  
    }  
  
    free(x_padded);  
    free(h_padded);  
    free(y_freq);  
    return len;  
}
```

## 计算多项式在多点的值

通过Chirp-Z变换：
$$P(x_j)=\sum_{k = 0}^{n - 1}a_k x_j^k=x_j^{-k_0^2/2}\sum_{k = 0}^{n - 1}(a_k x_j^{k_0^2/2})x_j^{-(k - k_0)^2/2}
$$
从而可以变为卷积
代码：
```c
void mul(const complex double a[], int n,  
         const complex double x[], int m,  
         complex double results[]) {  
    int L = 1;  
    while (L < n + m - 1) L <<= 1;  
    complex double *A = calloc(L, sizeof(complex double));  
    complex double *B = calloc(L, sizeof(complex double));  
    complex double *C = calloc(L, sizeof(complex double));  
   
    for (int k = 0; k < n; k++) {  // 构造Chirp-Z变换的卷积核 
        A[k] = a[k] * cexp(-I * PI * k * k / n);  
    }  
    for (int j = 0; j < m; j++) {  
        B[j] = cexp(I * PI * j * j / n) / x[j];  
    }  
    fft(A, L, 0);  
    fft(B, L, 0);  
    for (int i = 0; i < L; i++) C[i] = A[i] * B[i];  
    fft(C, L, 1);  
      
    for (int j = 0; j < m; j++) {  
        results[j] = C[j] * cexp(-I * PI * j * j / n);  
    }  
    free(A);  
    free(B);  
    free(C);  
}
```

```c
void multi(const complex double a[], int n,  
           complex double x0, complex double q, int m,  
           complex double results[]) {  
    int L = 1;  
    while (L < 2 * n) L <<= 1;  
    complex double *A = calloc(L, sizeof(complex double));  
    complex double *B = calloc(L, sizeof(complex double));  
  
    // 构造序列  
    complex double q_power = 1.0;  
    for (int k = 0; k < n; k++) {  
        A[k] = a[k] * cpow(x0, k) * cpow(q, -k*(k-1)/2);  
        q_power *= q;  
    }  
  
    for (int j = -n+1; j < m; j++) {  
        int idx = (j + L) % L;  
        B[idx] = cpow(q, j*(j-1)/2);  
    }  
    fft(A, L, 0);  
    fft(B, L, 0);  
    for (int i = 0; i < L; i++) A[i] = A[i] * B[i];  
    fft(A, L, 1);  
      
    for (int j = 0; j < m; j++) results[j] = A[j] * cpow(q, j*(j-1)/2);  
    free(A);  
    free(B);  
}
```
