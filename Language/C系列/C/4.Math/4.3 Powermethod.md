
幂方法适用于求大型稀疏矩阵的主特征值向量，时间复杂度为$O(kN^2)$

## 常数定义
```
#include <stdio.h>  
#include <stdlib.h>  
#include <math.h>  
#include <time.h>  
#include <string.h>
#define N 3       // 矩阵维度  
#define MAX_ITER 1000 // 最大迭代次数  
#define TOL 1e-6     // 收敛容忍度
```

## 矩阵-向量乘法: result = A * vec
```c
void mult(const double A[N][N],const double vec[N],double result[N]) {  
    for (int i = 0; i < N; i++) {  
        result[i] = 0.0;  
        for (int j = 0; j < N; j++) result[i] += A[i][j] * vec[j];  
    }  
}
```

## 计算向量的2-范数
```c
double norm(const double vec[N]) {  
    double sum = 0.0;  
    for (int i = 0; i < N; i++) sum += vec[i] * vec[i];  
    return sqrt(sum);  
}
```

## 主函数
```c
void power(double A[N][N], double vector[N], double* value) {  
    double v_old[N], v_new[N];  
    double lambda_old = 0.0, lambda_new; 
    srand(time(NULL));  //随机初始向量
    for(int i = 0; i < N; i++) v_old[i] = (double)rand()/RAND_MAX;  
    double nrm = norm(v_old);  
    for (int i = 0; i < N; i++) v_old[i] /= nrm;   
    for (int iter = 0; iter < MAX_ITER; iter++) { 
        mult(A, v_old, v_new);   
        lambda_new = 0.0;  
        for (int i = 0; i < N; i++) lambda_new += v_old[i]*v_new[i];  
        nrm = norm(v_new);  
        for (int i = 0; i < N; i++) v_new[i] /= nrm; 
        if (fabs(lambda_new - lambda_old) < TOL) break;  
        lambda_old = lambda_new;  
        memcpy(v_old,v_new,N*sizeof(double));  
    }  
    *value = lambda_new;  
    memcpy(vector,v_new,N*sizeof(double));  
}
```
