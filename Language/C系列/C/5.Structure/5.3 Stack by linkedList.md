
链表实现的栈可用于动态栈，定义为：
```c
typedef struct Node {  
    int data;  
    struct Node* next;    
} Node;  
typedef struct {  
    Node* top;  
}Stack;
```

## 初始化栈  
```c
void sinit(Stack* stack) {  
    stack->top = NULL;  
}  
```

## 判断空栈  
```c
int sisempty(Stack* stack) {  
    return stack->top == NULL;  
}  
```
## 压栈  
```c
int spush(Stack* stack, int value) {  
    Node* newNode = (Node*)malloc(sizeof(Node));  
    if (newNode == NULL) return 0;  
    newNode->data = value;  
    newNode->next = stack->top;  
    stack->top = newNode;  
    return 1;  
}  
```

## 弹栈  
```c
int spop(Stack* stack, int* value) {  
    if (sisempty(stack)) return 0;  
    Node* temp = stack->top;  
    *value = temp->data;  
    stack->top = temp->next;  
    free(temp);  
    return 1;  
}  
```

## 查看栈顶  
```c
int speek(Stack* stack, int* value) {  
    if (sisempty(stack)) return 0;  
    *value = stack->top->data;  
    return 1;  
}  
```

## 遍历打印栈内容（从栈顶到栈底）  
```c
int sprint(Stack* stack) {  
    if (sisempty(stack)) return 0;  
    printf("content: ");  
    Node* p = stack->top;  
    while (p != NULL) {  
        printf("%d ", p->data);  
        p = p->next;  
    }  
    printf("\n");  
    return 1;  
}  
```

## 释放整个栈  
```c
void sfree(Stack* stack) {  
    while (stack->top != NULL) {  
        Node* temp = stack->top;  
        stack->top = stack->top->next;  
        free(temp);  
    }  
}
```
