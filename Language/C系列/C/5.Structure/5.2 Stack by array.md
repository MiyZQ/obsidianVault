
数组实现的栈适用于静态栈
定义为：
```c
#define MAX_SIZE 100
typedef struct {  
    int data[MAX_SIZE];  
    int top;            // 初始为-1表示空栈  
} Stack;
```

## 初始化栈  
```c
void sinit(Stack *s) {  
    s->top = -1;  
}  
```

## 判断空栈  
```c
int sisempty(Stack *s) {  
    return s->top == -1;  
}  
```

## 判断满栈  
```c
int sisfull(Stack *s) {  
    return s->top == MAX_SIZE - 1;  
}  
```

## 压栈  
```c
int spush(Stack *s, int value) {  
    if (sisfull(s)) return 0;  
    s->data[++s->top] = value;  
    return 1;  
} 
``` 

## 弹栈  
```c
int spop(Stack *s) {  
    if (sisempty(s)) return -1;  
    return s->data[s->top--];  
}  
```

## 查看栈顶  
```c
int speek(Stack *s) {  
    if (sisempty(s)) return -1;  
    return s->data[s->top];  
}
```
