
一般默认为循环队列，此时可以解决假满问题，front指向首元素，rear指向空位置，位于尾元素后方

静态队列定义为：
```c
#define MAX_SIZE 100 // 定义队列的最大容量,实际可用max-1  
typedef struct {  
    int data[MAX_SIZE];  
    int front;  
    int rear;  
} Queue;
```

## 初始化队列  
```c
void qinit(Queue *q) {  
    q->front = 0;  
    q->rear = 0;  
}  
```
  
## 判断空队列 
```c
bool qisempty(Queue *q) {  
    return q->front == q->rear;  
}  
```
  
## 判断满队列
```c
bool qisfull(Queue *q) {  
    return (q->rear + 1) % MAX_SIZE == q->front;  
}  
```
  
## 入队  
```c
bool qen(Queue *q, int value) {  
    if (qisfull(q)) return false;  
    q->data[q->rear] = value;  
    q->rear = (q->rear + 1) % MAX_SIZE;  
    return true;  
}  
```
  
## 出队  
```c
bool qde(Queue *q, int *value) {  
    if (qisempty(q)) return false;  
    *value = q->data[q->front];  
    q->front = (q->front + 1) % MAX_SIZE;  
    return true;  
}  
```
  
## 获取队首元素(不删除)  
```c
bool qfront(Queue *q, int *value) {  
    if (qisempty(q)) return false;  
    *value = q->data[q->front];  
    return true;  
}  
```
  
## 打印队列元素  
```c
void qprint(Queue *q) {  
    if (qisempty(q)) { 
	puts("None");
	return;
	}  
    printf("content: ");  
    int i = q->front;  
    while (i != q->rear) {  
        printf("%d ", q->data[i]);  
        i = (i + 1) % MAX_SIZE;  
    }  
    printf("\n");  
}  
```
  
## 清空队列(快速方法)  
```c
void qclear(Queue *q) {  
    q->front = 0;  
    q->rear = 0;  
}
```
