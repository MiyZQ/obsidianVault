
链表的结构体定义为：
```c
typedef struct Node {  
    int data;          
    struct Node* next;
} Node;
```

关于链表操作函数：
	以下采用返回值函数，调用需要重新赋值`head = fun(head,...)`,是为了处理空表这一特殊情况(刷新原空的头指针)  
	若采用无返回值函数需要接收二级指针`Node**head`

## 创建新节点
```c
Node* create(int data) {  
    Node* newNode = (Node*)malloc(sizeof(Node));  
    if (newNode == NULL) return NULL;  
    newNode->data = data;  
    newNode->next = NULL;  
    return newNode;  
}
```

## 在链表头部插入节点
```c
Node* inH(Node* head, int data) {  
    Node* newNode = create(data);  
    newNode->next = head;  
    return newNode; 
}
```

## 在链表尾部插入节点
```c
Node* inT(Node* head, int data) {  
    Node* newNode = create(data);  
    if (head == NULL) return newNode;  // 空链表，新节点就是头节点  
    Node* p = head;  
    while (p->next != NULL) p = p->next;  
    p->next = newNode;  
    return head;  
}
```

## 插入指定位置
```c
Node* insert(Node* head, int data, int position) {  
    if (position <= 0) return inH(head,data);    
    Node* newNode = create(data);  
    Node* p = head;  
    for (int i = 0; p->next != NULL && i < position - 1; i++) p = p->next;   
    newNode->next = p->next;  
    p->next = newNode;  
    return head;  
}
```

## 删除指定数据的节点
```c
Node* ldel(Node* head, int data) {  
    if (head == NULL) return NULL; 
    if (head->data == data) {  
        Node* temp = head->next;  
        free(head);  
        return temp;  
    }  
    Node* p = head;  
    while (p->next != NULL && p->next->data != data) p = p->next;  
    if (p->next == NULL) {
	    puts("Not found");
	    return head; 
	}
    Node* temp = p->next;  
    p->next = temp->next;  
    free(temp);  
    return head;  
}
```

## 查找指定数据的节点
```c
Node* lsearch(Node* head, int data) {  
    Node* p = head;  
    while (p != NULL) {  
        if (p->data == data) {  
            return p;  
        }  
        p = p->next;  
    }  
    return NULL;  
}
```

## 获取链表长度
```c
int llen(Node* head) {  
    int length = 0;  
    Node* p = head;  
    while (p != NULL) {  
        length++;  
        p = p->next;  
    }  
    return length;  
}
```

## 遍历打印链表
```c
void lprint(Node* head) {  
    Node* p = head;  
    printf("content: ");  
    while (p != NULL) {  
        printf("%d -> ", p->data);  
        p = p->next;  
    }  
    printf("NULL\n");  
}
```

## 释放整个链表
```c
void lfree(Node* head) {  
    Node* p = head;  
    Node* next;  
    while (p != NULL) {  
        next = p->next;  
        free(p);  
        p = next;  
    }  
}
```
