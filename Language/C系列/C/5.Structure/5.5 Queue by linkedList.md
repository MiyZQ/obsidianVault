
动态队列没有假满问题，此时不需要用循环队列解决，rear指向尾元素

动态队列定义为：
```c
typedef struct Node {  
    int data;  
    struct Node* next;  
} Node;  
typedef struct {  
    Node* front;  
    Node* rear;  
    int size;  
} Queue;
```

## 初始化队列
```c
void qinit(Queue* q) {  
    q->front = NULL;  
    q->rear = NULL;  
    q->size = 0;  
}  
```
  
## 判断空队列
```c
bool qisempty(Queue* q) {  
    return q->front == NULL;  
}  
```
  
## 入队  
```c
bool qen(Queue* q, int value) {  
    Node* newNode = (Node*)malloc(sizeof(Node));  
    if (!newNode) return false;  
    newNode->data = value;  
    newNode->next = NULL;  
    if (qisempty(q)) {  
        q->front = newNode;  
        q->rear = newNode;  
    } else {  
        q->rear->next = newNode;  
        q->rear = newNode;  
    }  
    q->size++;  
    return true;  
}  
```
  
## 出队  
```c
bool qde(Queue* q,int *value) {  
    if (qisempty(q)) return false;  
    Node* temp = q->front;  
    *value = temp->data;  
    q->front = q->front->next;  
    free(temp);  
    if (q->front == NULL) q->rear = NULL;  
    q->size--;  
    return true;  
}  
```
  
## 获取队首元素(不删除)  
```c
bool qfront(Queue* q,int *value) {  
    if (qisempty(q)) return false;  
    *value = q->front->data;  
    return true;  
}  
```
  
## 读取队列长度  
```c
int qlen(Queue* q) {  
    return q->size;  
}
```  
  
## 若不便储存size，可用遍历法读取长度  
```c
int qlen(Queue* q) {  
    int count = 0;  
    Node* p = q->front;  
    while (p != NULL) {  
        count++;  
        p = p->next;  
    }  
    return count;  
}  
```
  
## 释放整个队列  
```c
void qclear(Queue* q) {  
    int tmp;  
    while (!qisempty(q)) qde(q,&tmp);  
}  
```
  
## 也可遍历释放  
```c
void qclear(Queue* q) {  
    Node* p = q->front;  
    while (p != NULL) {  
        Node* temp = p;  
        p = p->next;  
        free(temp);  
    }  
    q->front = NULL;  
    q->rear = NULL;  
    q->size = 0;  
}  
```
如果队列结构本身是动态分配的，也需要释放q本身
  
## 打印队列元素  
```c
bool qprint(Queue* q) {  
    if (qisempty(q)) return false;  
    printf("content: ");  
    Node* p = q->front;  
    while (p != NULL) {  
        printf("%d ", p->data);  
        p = p->next;  
    }  
    printf("\n");  
    return true;  
}
```
