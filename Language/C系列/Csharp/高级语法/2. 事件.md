
C# 中事件是一种成员，使用事件来进行线程间的通信

==发布器类==即包含事件的类，用于发布事件
==订阅器类==即接受事件的类

声明事件
```cs
public delegate void A(string status);
public event A EventLog;
```

> [!Attention]
> 触发事件时，最好使用 `?.Invoke()` 语法确保只有在有订阅者时才调用事件

用例
```cs
using System;
namespace MyEvent
{
    // 发布器类
    public class EventTest
    {
        private int value;
        public delegate void Handler();
        public event Handler? ChangeValue;
        protected virtual void OnValueChanged()
        {
            if (ChangeValue is not null)
            {
                ChangeValue();  
            }
            else
            {
                Console.WriteLine("event not fire");
                Console.ReadKey();
            }
        }
        public EventTest(int val=1)
        {
            SetValue(val);
        }
        public void SetValue(int val)
        {
            if (value != val)
            {
                value = val;
                OnValueChanged();
            }
        }
    }
    // 订阅器类
    public class SubScribeEvent
    {
        public void print()
        {
            Console.WriteLine("event fire");
        }
    }
    // 触发
    public class MainClass
    {
        public static void Main()
        {
            var e = new EventTest();
            var v = new SubScribeEvent();
            e.ChangeValue += new EventTest.Handler(v.print);
            e.SetValue(7);
            e.SetValue(11);
        }
    }
}
```
