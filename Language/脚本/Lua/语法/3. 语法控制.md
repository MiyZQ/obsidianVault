
### 循环

```lua
-- while 循环
while true do
    print(1)
    break
end

-- 数值循环 step默认1
for var=start,stop,step do
    print(var)
end

-- 泛型循环
a = {'1','2','3'}
for i,v in ipairs(a) do
    print(i,v)
end

-- repeat 循环
repeat
    a = a+1
    print(1)
until a>15

-- goto 语句用于实现continue
if a>15 then
	goto label
end
::label:: print("yes")
```

### 条件

```lua
if 0
then
    print(1)
elseif nil
then
    print(0)
else
    print(1)
end
```
